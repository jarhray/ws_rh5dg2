cmake_minimum_required(VERSION 3.8)
project(rh5dg2_driver)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rh5dg2_interfaces REQUIRED)
find_package(Threads REQUIRED)
add_library(rh5dg2_core src/core.cpp)
target_link_libraries(rh5dg2_core PUBLIC Threads::Threads)
target_include_directories(rh5dg2_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_compile_options(rh5dg2_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hand_node src/hand_node.cpp)
target_link_libraries(hand_node rh5dg2_core)
ament_target_dependencies(hand_node rclcpp sensor_msgs std_srvs rh5dg2_interfaces)
target_compile_options(hand_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS rh5dg2_core hand_node EXPORT rh5dg2_coreTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_core test/test_core.cpp)
  target_link_libraries(test_core rh5dg2_core Threads::Threads)
  add_test(NAME protocol_core COMMAND test_core)
  set_tests_properties(protocol_core PROPERTIES TIMEOUT 30)
  add_executable(test_registers test/test_registers.cpp)
  target_include_directories(test_registers PRIVATE include)
  add_test(NAME command_registers COMMAND test_registers)
endif()
ament_export_targets(rh5dg2_coreTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Threads)
ament_export_include_directories(include)
ament_package()
